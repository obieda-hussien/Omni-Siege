cmake_minimum_required(VERSION 3.22.1)
project(OmniSiegeNative LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(ANDROID)
    add_library(omni_siege SHARED siege_engine.cpp siege_jni.cpp)
    target_compile_options(omni_siege PRIVATE -Wall -Wextra -Werror)
else()
    enable_testing()
    add_executable(siege_native_tests siege_engine.cpp siege_engine_tests.cpp)
    target_compile_options(siege_native_tests PRIVATE -Wall -Wextra -Werror)
    add_test(NAME siege_native_tests COMMAND siege_native_tests)
endif()
